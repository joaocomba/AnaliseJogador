% --- SLIDE 6: MODELO DE VALORAÇÃO DE MERCADO ---
\begin{frame}{6. Modelo de Precificação: Valor de Desempenho (€)}
  \begin{block}{Conceito do Modelo}
    O \textbf{Valor de Desempenho (€)} ajusta o valor de mercado real do Sofascore cruzando a qualidade estatística na posição, a amostragem de jogos e a idade do atleta.
  \end{block}

  \begin{equation*}
    \text{Valor de Desempenho (€)} = \text{Valor Base (€)} \times \mathbf{M_{\text{desempenho}}} \times \mathbf{W_{\text{amostra}}} \times \mathbf{M_{\text{idade}}}
  \end{equation*}

  \begin{itemize}
    \item \textbf{Multiplicador de Desempenho ($M_{\text{desempenho}}$)}: Calculado pelo percentil médio do jogador ($0.60\times$ a $1.40\times$) nas estatísticas específicas da sua posição (incluindo perfil próprio para \texttt{LD/LE}).
    \item \textbf{Novas Colunas Geradas}:
      \begin{enumerate}
        \item \textbf{Diferença de Valor (€)}: $\text{Valor de Desempenho} - \text{Valor Real}$
        \item \textbf{Diferença (\%)}: $\frac{\text{Valor de Desempenho} - \text{Valor Real}}{\text{Valor Real}} \times 100$
      \end{enumerate}
  \end{itemize}
\end{frame}

% --- SLIDE 7: FATORES DE AMOSTRAGEM E IDADE ---
\begin{frame}{7. Fatores de Ajuste: Amostragem e Idade}
  \begin{columns}
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Fator de Amostragem ($W_{\text{amostra}}$)}
        Evita distorções por amostra pequena (poucos jogos):
        \begin{equation*}
          W_{\text{amostra}} = \text{clip}\left(\frac{\text{Partidas}}{15}, \, 0.20, \, 1.00\right)
        \end{equation*}
        \begin{itemize}
          \item $1 - 3$ jogos: Peso $\sim 20\% - 30\%$ (ancorado no valor base).
          \item $15+$ jogos: Peso de $100\%$ (confiança estatística total).
        \end{itemize}
      \end{block}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{block}{2. Fator de Idade e Potencial ($M_{\text{idade}}$)}
        Reflete o potencial futuro de revenda:
        \begin{itemize}
          \item \textbf{$\le 23$ anos}: $+5\%$ ao ano de juventude.
            \begin{equation*}
              M_{\text{idade}} = 1.0 + 0.05 \times (23 - \text{Idade})
            \end{equation*}
          \item \textbf{$24 - 28$ anos}: Ápice físico/tático ($1.00\times$).
          \item \textbf{$> 28$ anos}: Depreciação de $-3.5\%$ ao ano.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% --- SLIDE 8: MOTOR DE SIMILARIDADE E FILTROS LATERAIS ---
\begin{frame}{8. Motor de Similaridade, Pesos \& Filtros}
  \begin{block}{Distância Euclidiana Ponderada nos Percentis}
    Permite encontrar os \textbf{Gêmeos Estatísticos} do jogador alvo usando métricas e pesos selecionados ($w_i \in [0.1\times, 3.0\times]$):
  \end{block}

  \begin{equation*}
    d(j, \text{alvo}) = \sqrt{\sum_{i=1}^{k} w_i \cdot \left( P_{j, i} - P_{\text{alvo}, i} \right)^2}
  \end{equation*}

  \begin{itemize}
    \item \textbf{Predefinições de Perfil}: Seleção automática para \textit{Ataque}, \textit{Meio-Campo}, \textit{Laterais}, \textit{Zaga/Volante}, \textit{Goleiro} ou \textit{Customizado}.
    \item \textbf{Filtros Interativos na Sidebar}:
      \begin{itemize}
        \item \textbf{Faixa de Idade}: Slider duplo para limitar anos de nascimento.
        \item \textbf{Valor de Mercado (€M)}: Caixas de entrada numéricas digitáveis ($\text{Mín}$ e $\text{Máx}$).
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{10. Conclusão e Aplicações Práticas}
  \begin{alertblock}{Aplicações para Scouting e Análise de Mercado}
    \begin{itemize}
      \item \textbf{Identificação de Pechinchas (Under-valued)}: Filtragem de jogadores onde o $\text{Valor de Desempenho} \gg \text{Valor Real}$.
      \item \textbf{Busca por Substitutos}: Encontrar substitutos estatísticos diretos para atletas negociados.
      \item \textbf{Ponderação Especializada por Posição}: Separação entre laterais e zagueiros para análises táticas fiéis.
      \item \textbf{Filtros de Entrada Numérica}: Pesquisas direcionadas por teto orçamentário digitável em milhões de euros.
